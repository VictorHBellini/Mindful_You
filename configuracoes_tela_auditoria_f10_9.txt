class _ConfiguracaoTelaState extends State<ConfiguracaoTela> {
  bool notificacao = false;
  bool temaEscuro = false;
  bool _exclusaoEmAndamento = false;

  String nomeUsuario = "UsuÃ¡rio";
  String emailUsuario = "";

  @override
  void initState() {
    super.initState();
    carregarDados();
  }

  Future<void> carregarDados() async {
    final prefs = await SharedPreferences.getInstance();

    if (!mounted) return;

    setState(() {
      nomeUsuario = prefs.getString('nomeUsuario') ?? "UsuÃ¡rio";
      emailUsuario = prefs.getString('emailUsuario') ?? "";
      notificacao = prefs.getBool('notificacao') ?? false;
      temaEscuro = prefs.getBool('temaEscuro') ?? false;
    });
  }

  Future<void> salvarPreferencias() async {
    final prefs = await SharedPreferences.getInstance();

    await prefs.setBool(
      'notificacao',
      notificacao,
    );

    await prefs.setBool(
      'temaEscuro',
      temaEscuro,
    );
  }

  Future<String?> _solicitarSenhaExclusao() async {
    final controlador = TextEditingController();

    try {
      return await showDialog<String>(
        context: context,
        barrierDismissible: false,
        builder: (dialogContext) => AlertDialog(
          title: const Text('Confirmar identidade'),
          content: Column(
            mainAxisSize: MainAxisSize.min,
            children: [
              const Text(
                'Digite sua senha local para confirmar '
                'a exclusÃ£o dos dados deste dispositivo.',
              ),
              const SizedBox(height: 16),
              TextField(
                controller: controlador,
                obscureText: true,
                autofocus: true,
                decoration: const InputDecoration(
                  labelText: 'Senha',
                  border: OutlineInputBorder(),
                ),
              ),
            ],
          ),
          actions: [
            TextButton(
              onPressed: () => Navigator.pop(dialogContext),
              child: const Text('Cancelar'),
            ),
            TextButton(
              onPressed: () {
                Navigator.pop(dialogContext, controlador.text);
              },
              child: const Text('Confirmar'),
            ),
          ],
        ),
      );
    } finally {
      controlador.dispose();
    }
  }

  Future<void> _excluirDadosLocais() async {
    if (_exclusaoEmAndamento || !mounted) return;

    setState(() => _exclusaoEmAndamento = true);
    var contaExcluida = false;

    try {
      final confirmar = await showDialog<bool>(
        context: context,
        barrierDismissible: false,
        builder: (dialogContext) => AlertDialog(
          title: const Text('Excluir dados locais?'),
          content: const Text(
            'Sua conta e seus histÃ³ricos salvos neste dispositivo '
            'serÃ£o excluÃ­dos permanentemente. '
            'Esta aÃ§Ã£o nÃ£o exclui contas ou dados do Firebase.',
          ),
          actions: [
            TextButton(
              onPressed: () => Navigator.pop(dialogContext, false),
              child: const Text('Cancelar'),
            ),
            TextButton(
              onPressed: () => Navigator.pop(dialogContext, true),
              style: TextButton.styleFrom(foregroundColor: Colors.red),
              child: const Text('Continuar'),
            ),
          ],
        ),
      );

      if (confirmar != true || !mounted) return;

      final sessao = await SessionService.validar();
      if (sessao == null || !mounted) return;

      final senha = await _solicitarSenhaExclusao();
      if (senha == null || senha.isEmpty || !mounted) return;

      final db = await DatabaseService.instance.database;

      final usuario = await db.query(
        'usuarios',
        columns: ['foto_path'],
        where: 'id = ?',
        whereArgs: [sessao.id],
        limit: 1,
      );

      if (!mounted || usuario.length != 1) return;

      final coordinator = ExclusaoLocalCoordinator(
        sessao: SessaoLocalAdapter(),
        senha: SenhaLocalAdapter(),
        repositorio: RepositorioExclusaoLocalAdapter(),
      );

      final resultado = await coordinator.executar(
        senhaDigitada: senha,
        usuarioEsperadoId: sessao.id,
      );

      if (!mounted) return;

      switch (resultado) {
        case ResultadoExclusaoLocal.excluido:
          contaExcluida = true;
          break;

        case ResultadoExclusaoLocal.senhaIncorreta:
          ScaffoldMessenger.of(context).showSnackBar(
            const SnackBar(
              content: Text('Senha incorreta. Nenhum dado foi excluÃ­do.'),
            ),
          );
          return;

        case ResultadoExclusaoLocal.sessaoInvalida:
        case ResultadoExclusaoLocal.sessaoAlterada:
          ScaffoldMessenger.of(context).showSnackBar(
            const SnackBar(
              content: Text(
                'A sessÃ£o mudou ou expirou. Nenhum dado foi excluÃ­do.',
              ),
            ),
          );
          return;

        case ResultadoExclusaoLocal.contaNaoEncontrada:
          ScaffoldMessenger.of(context).showSnackBar(
            const SnackBar(
              content: Text('A conta local nÃ£o foi encontrada.'),
            ),
          );
          return;

        case ResultadoExclusaoLocal.vinculoOuOperacaoBloqueada:
          ScaffoldMessenger.of(context).showSnackBar(
            const SnackBar(
              content: Text(
                'ExclusÃ£o bloqueada: existe um vÃ­nculo Firebase registrado.',
              ),
            ),
          );
          return;
      }

      // Uma falha na limpeza da foto nÃ£o deve manter uma sessÃ£o de conta excluÃ­da.
      try {
        await FotoPerfilService.removerFotoAntiga(
          usuario.single['foto_path'] as String?,
        );
      } catch (_) {
        // Limpeza de arquivo secundÃ¡ria; a conta jÃ¡ foi excluÃ­da do SQLite.
      }

      await SessionService.encerrar();
      if (!mounted) return;

      Navigator.pushNamedAndRemoveUntil(context, '/login', (_) => false);
    } on StateError {
      if (!mounted) return;
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(
          content: Text(
            contaExcluida
                ? 'Os dados locais foram excluÃ­dos, mas nÃ£o foi possÃ­vel '
                    'finalizar a sessÃ£o. Reinicie o aplicativo.'
                : 'ExclusÃ£o bloqueada: hÃ¡ um vÃ­nculo Firebase registrado '
                    'ou a operaÃ§Ã£o nÃ£o pÃ´de ser concluÃ­da.',
          ),
        ),
      );
    } catch (_) {
      if (!mounted) return;
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(
          content: Text(
            contaExcluida
                ? 'Os dados locais foram excluÃ­dos, mas ocorreu uma falha '
                    'ao encerrar a sessÃ£o. Reinicie o aplicativo.'
                : 'NÃ£o foi possÃ­vel concluir a exclusÃ£o. '
                    'Verifique os dados e tente novamente.',
          ),
        ),
      );
    } finally {
      if (mounted) {
        setState(() => _exclusaoEmAndamento = false);
      }
    }
  }
